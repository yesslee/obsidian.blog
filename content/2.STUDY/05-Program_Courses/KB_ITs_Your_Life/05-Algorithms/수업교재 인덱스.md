```dataviewjs
const pages = dv.pages('"2.STUDY/05-Program_Courses/KB_ITs_Your_Life/05-Algorithms/수업교재"')
    .where(p => p.weeks)
    .groupBy(p => p.weeks)
    .sort(g => g.key, 'asc');

// 주차별 row 생성
let tableRows = pages.map(group => {
    let week = `${group.key}주차`;
    let notes = group.rows
    .sort(p => p.order, 'asc')
    .map(p => `[[${p.file.path}|${p.title}]]`)
    .join(" / ");
    return [week, notes];
});

// 테이블 출력
dv.table(["주차", "노트"], tableRows);
```
