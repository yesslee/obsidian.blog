```java
// 정확성 X & 시간초과 코드
import java.lang.Math;
class Solution {
    public int[] solution(int[] sequence, int k) {
        int[] answer = {0, 0};
        int start = 0;
        int end = sequence.length - 1;
        int len = sequence.length - 1;
        
        for (int j = end; j >= 0; j--) {
            for (int i = start; i <= j; i++) {
                // System.out.println("[" + sequence[i] + ", " + sequence[j] + "]");
                int sum = 0;
                for (int h = i; h <= j; h++) {
                    sum += sequence[h];
                    if (sum == k) {
                        // System.out.println(sum);
                        len = Math.min(len, j - i);
                        answer[0] = i;
                        answer[1] = j;
                        // System.out.println(j - i);
                    }
                }
            }
        }
        
        // System.out.println(len);
        
        return answer;
    }
}

// 슬라이딩 윈도우 구현 코드
class Solution {
    public int[] solution(int[] sequence, int k) {
        int[] answer = {0, 1000000};
        int start = 0;
        int sum = 0;
        
        for (int end = 0; end < sequence.length; end++) {
            sum += sequence[end];
            while (sum > k) {
                sum -= sequence[start];
                start++;
            }
            
            if (sum == k) {
                if ((end - start) < (answer[1] - answer[0])) {
                    answer[0] = start;
                    answer[1] = end;
                }
            }
        }
        
        return answer;
    }
}
```
