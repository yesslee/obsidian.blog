#### 문제파악
#### 접근 방법
물품을 최대한 많은 부서에게 지원하기 위해서는 신청한 금액이 적은 부서부터 신청 금액을 더하며 부서의 개수를 count한 뒤, 신청 금액의 합이 예산과 같거나 넘어서기 전까지만 세면 되지 않을까 생각했다.
#### 코드 구현
```java
import java.util.Arrays;
class Solution {
    public int solution(int[] d, int budget) {
        Arrays.sort(d);
        int answer = 0;
        int sum = 0;
        
        for (int i = 0; i < d.length; i++) {
            sum += d[i];
            answer++;
            if (sum == budget) break;
            else if (sum > budget) {
                answer--;
                break;
            }
        }
                
        return answer;
    }
}
```

```java
// 생각해보니 주어진 예산에서 부서별 신청 금액을 빼는 게 더 맞는 것 같아서 약간 수정한 코드
import java.util.Arrays;
class Solution {
    public int solution(int[] d, int budget) {
        Arrays.sort(d);
        int answer = 0;
        int rest = budget;
        
        for (int i = 0; i < d.length; i++) {
            rest -= d[i];
            answer++;
            if (rest == 0) break;
            else if (rest < 0) {
                answer--;
                break;
            }
        }
                
        return answer;
    }
}
```

import java.util.Arrays;
class Solution {

            if (rest < 0) break;
