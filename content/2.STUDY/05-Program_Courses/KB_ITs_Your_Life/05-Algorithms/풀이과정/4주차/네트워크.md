#### 문제파악
컴퓨터의 개수 = n
#### 접근 방법
컴퓨터의 개수만큼 BFS를 실행

노드 간 연결성 판단 = BFS, DFS 사용
visited로 방문여부 체크 → 방문 안 한 노드라면 DFS 진행
DFS 할 때 마다 카운트 하면 그게 네트워크의 갯수가 된다

네트워크가 여러개 있다면 bfs/dfs 이용해서 방문여부를 체크하고, 방문 안 했다면 bfs/dfs 실행하여 카운트

// 혼자 구현 시도해본 코드(완성 X)
import java.util.*;

class Solution {
    static Map<Integer, List<Integer>> graph = new HashMap<>();
    static Map<Integer, Boolean> visited = new HashMap<>();
    
    public void bfs(int startVertex) {
        Queue<Integer> queue = new LinkedList<>();
        queue.offer(startVertex);
        visited.put(startVertex, true);
        
        while (!queue.isEmpty()) {
            int currVertex = queue.poll();
            System.out.print(currVertex + " ");
            for (int nextVertex : graph.get(currVertex)) {
                if (!visited.containsKey(nextVertex)) {
                    queue.offer(nextVertex);
                    visited.put(nextVertex, true);
                }
            }
        }
    }
    
    public static void makeGraph(int[][] computers) {
        for (int i = 0; i < computers.length; i++) {
            graph.put(i, new ArrayList<>());
            for (int j = 0; j < computers[i].length; j++) {
                if (computers[i][j] == 1 && i != j) {
                    graph.get(i).add(j);
                }
            }
        }
    }
    
    public int solution(int n, int[][] computers) {
        int answer = 0;
        
        makeGraph(computers);
        
        for (int i = 0; i < n; i++) {
			      bfs(i);
			      answer++;
	      }
    
        return answer;
    }    
}

// 해설 듣고 수정해서 완성한 코드(BFS)
import java.util.*;

class Solution {
    boolean[] visited;
    public int solution(int n, int[][] computers) {
        visited = new boolean[n];
        int answer = 0; // 네트워크의 개수

        for (int i = 0; i < n; i++) {
            if (visited[i]) continue;
            bfs(n, computers, i);
            answer++;
        }
        return answer;
    }

    public void bfs(int n, int[][] computers, int startVertex) {
        Queue<Integer> queue = new LinkedList<>();
        queue.offer(startVertex);
        visited[startVertex] = true;

        while (!queue.isEmpty()) {
            int currVertex = queue.poll();
            for (int i = 0; i < n; i++) {
                if (!visited[i] && computers[currVertex][i] == 1) {
                    queue.offer(i);
                    visited[i] = true;
                }
            }
        }
    }
}

// DFS로 구현한 코드(해설)
import java.util.*;

class Solution {
		boolean[] visited;
		public int solution(int n, int[][] computers) {
				int answer = 0;
				visited = new boolean[n];
			
				for (int i = 0; i < n; i++) {
						if (!visited[i]) {
								dfs(n, computers, i);
								answer++;
						}
				}
				return answer;
		}
		
		void dfs(int n, int[][] computers, int curr) {
				visited[curr] = true;
				
				for (int i = 0; i < n; i++) {
						if (!visited[i] && computers[curr][i] == 1) {
								dfs(n, computers, i);
						}
				}
		}
}
