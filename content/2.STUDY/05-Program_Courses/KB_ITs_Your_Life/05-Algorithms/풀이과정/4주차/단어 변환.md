#### 문제파악
주어진 begin 문자열을 한 글자씩만 변경하되, words 배열에 존재하는 문자열로만 변경가능할 때 target 문자열로 변경하기까지 필요한 **최소 횟수** 구하기
→ [[2.STUDY/05-Program_Courses/KB_ITs_Your_Life/05-Algorithms/풀이과정/4주차/Coin Change|Coin Change]] 문제와 유사하다!
#### 접근 방법
인접한 곳으로 이동하며 탐색하여 최소 횟수 구하는 문제
→ BFS로 구현하자

문자 차이 몇개인지 확인하는 함수 따로 구현해서 문자 차이가 1개인 경우만 nextVertex로 예약 큐에 넣기

예약할 때 몇 번째만에 왔는지 같이 기록해줘야 변경 횟수 구할 수 있음

1. 달라진 글자 수 확인하는 함수 구현
2. 큐에 문자열과 이동 횟수 같이 기록
	→ 어떻게 큐에 문자열하고 이동횟수 같이 넣지???
	
	코치님 힌트
	1. 새로운 클래스를 만들어서 큐에 넣자
	2. 애초에 글자 수 차이가 1인 애들만 체크해서 큐에 넣자
	3. ArrayList에 제네릭 사용 안하면 아무거나 넣을 수 있으니 그렇게 넣자

한글자 차이 나는 단어들로 그래프가 생성 → 인접한 노드 먼저 탐색 = bfs
#### 코드 구현
```java
// 혼자 구현 시도해본 코드(완성 X)
import java.util.*;

public class Solution {
    boolean[] visited;
    public int solution(String begin, String target, String[] words) {
        visited = new boolean[words.length];
        int answer = 0;
        String[] newWords = diff(begin, words);

        for (int i = 0; i < newWords.length; i++) {
            bfs(i, newWords);
        }

        return answer;
    }

    void bfs(int startVertex, String[] words) {
        Queue<Integer> queue = new LinkedList<>();
        queue.offer(startVertex);
        visited[startVertex] = true;

        while (!queue.isEmpty()) {
            int currVertex = queue.poll();
            for (int i = 0; i < words.length; i++) {
                if (!visited[i]) {
                    queue.offer(i);
                    visited[i] = true;
                }
            }
        }

    }
		
    private String[] diff(String current, String[] words) {
        List<String> wordList = new ArrayList<>();

        for (int i = 0; i < words.length; i++) {
            int count = 0;
            for (int j = 0; j < current.length(); j++) {
                if (current.charAt(i) != words[i].charAt(j)) {
                    count++;
                }
            }
            if (count == 1) {
                wordList.add(current);
            }
        }

        String[] newWords = new String[wordList.size()];
        for (int i = 0; i < wordList.size(); i++) {
            newWords[i] = wordList.get(i);
        }

        return newWords;
    }
}
```

```java
// 해설 코드
import java.util.*;

public class Solution {
	public int solution(String begin, String target, String[] words) {
		Queue<WordState> queue = new ArrayDeque<>(); // 문자열과 이동 횟수 같이 기록 필요
		boolean[] visited = new boolean[words.length];

		queue.offer(new WordState(begin, 0));

		while (!queue.isEmpty()) {
			// 방문
			WordState curr = queue.poll();
			// 방문한 곳이 내가 찾던 target이라면 return count
			if (curr.word.equals(target)) return curr.count;
			
			// nextVertex 예약 (begin 문자열과 한 문자만 다른 words 배열 내의 문자열)
			for (int i = 0; i < words.length; i++) {
				if (getDiffCnt(curr.word, words[i]) == 1) {
					if (!visited[i]) {
						queue.offer(new WordState(words[i], curr.count + 1));
						visited[i] = true;
					}
				}
			}
		}
		
		int answer = 0;
		return answer;
  }
  
  int getDiffCnt(String word, String target) {
	  int diffCount = 0;
	  for (int i = 0; i < word.length(); i++) {
		  if (word.charAt(i) != target.charAt(i)) diffCount++;
	  }
	  
	  return diffCount;
  }
  
  // 큐에 값 2개 넣기 위한 클래스 생성
	class WordState {
		String word;
		int count;
		
		WordState(String word, int count) {
			this.word = word;
			this.count = count;
		}
	}
}
```

#### 배우게 된 점
DFS로 풀이 불가능한 이유
→ 문자열 변환에 필요한 최소 횟수 보장 불가

그래프의 DFS = 노드를 딱 1회만 방문
타겟 문자열 발견까지 걸린 횟수가 최소 횟수인지 알 수 없어짐

완전탐색으로도 풀 수 있다(완전탐색 ≠ DFS)
