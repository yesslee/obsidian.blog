트리 dfs로 탐색

다음 단계 넘어갈 때 합을 누적할 것 → curr 변수에 지금까지 누적된 합을 넣음

curr의 값이 target과 같을 때 count를 +1 하거나 1을 return해서 그 합을 구함

핵심은 dfs를 + 방향과 -방향으로 나눠서 진행해본다는 것!

basecase = index개수와 numbers개수 일치
#### 코드 구현
```java
// 구현 시도 코드
class Solution {
    int answer;
    int g_target;
    boolean[] visited;
    public int solution(int[] numbers, int target) {
        answer = 0;
        g_target = target;
        visited = new boolean[numbers.length];
        
        dfs(numbers, 0);
        
        return answer;
    }
    
    void dfs(int[] numbers, int curr) {
        visited[curr] = true;
        
        int sum = 0;
        for (int i = 0; i < numbers.length; i++) {
            if (!visited[i] && sum != g_target) {
                dfs(numbers, i);
            }
        }
    }
}
```

```java
// 완전 탐색 코드
public class Solution {
    int answer;
    public int solution(int[] numbers, int target) {
        answer = 0;
        backtrack(0, 0, numbers, target);

        return answer;
    }

    void backtrack(int curr, int idx, int[] numbers, int target) {
        // curr에 idx 값 더하거나 빼기
        // base case
        if (idx == numbers.length) {
            if (target == curr) answer++;
            return;
        }

        // recursive call
        backtrack(curr + numbers[idx], idx + 1, numbers, target);
        backtrack(curr - numbers[idx], idx + 1, numbers, target);
    }
}
```

// 해설 코드 (answer 전역변수 사용 X, return 활용)
class Solution {
		int g_target = target;
    public int solution(int[] numbers, int target) {
        return dfs(numbers, 0, 0)
    }
    
    int dfs(int[] numbers, int idx, int curr) {
	    if (idx == numbers.length) {
		    if (curr == g_target) {
			    return 1;
		    }
		    return 0;
	    }
	    
	    int sum = 0;
	    sum += dfs(numbers, idx + 1, curr + numbers[idx]);
	    sum += dfs(numbers, idx + 1, curr - numbers[idx]);
	    
	    return sum;
    }
}
