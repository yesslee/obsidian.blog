#### 문제파악
maps는 n * m 크기의 2차원 배열

maps는 0과 1로만 이루어져 있으며, 0 = 벽, 1 = 벽이 없는 자리

시작점 = 좌측 상단인 (1, 1) 위치

상대방 진영 = 우측 하단인 (n, m) 위치

동서남북 4방향 이동만 가능할 때, 상대방 진영에 최대한 빨리 도착하기위해 지나가야 하는 칸의 갯수를 return

도착할 수 없을 때는 -1 return
#### 접근 방법
최단거리 이동 = BFS로 구현하면 된다!!

대신 문제에 따라 조건문 작성할 때 유의할 것

// 처음 구현한 코드(테스트케이스 2번에서 메모리 초과 발생)
// -> isValid() 체크하고선 visited 체크 안했기 때문에 발생한 문제!!!
import java.util.*;
class Solution {
    static int rowLength, colLength;
    boolean[][] visited;
    int[] dr = {1, 0, -1, 0};
    int[] dc = {0, 1, 0, -1};
    
    public boolean isValid(int r, int c, int[][] maps) {
        return (0 <= r && r < rowLength) && (0 <= c && c < colLength) && (maps[r][c] == 1);
    }
    
    public int solution(int[][] maps) {
        int answer = -1;
        rowLength = maps.length;
        colLength = maps[0].length;
            
        visited = new boolean[rowLength][colLength];
        
        if (maps[0][0] != 1 || maps[rowLength - 1][colLength - 1] != 1) {
            return answer;
        }
        
        Queue<int[]> queue = new ArrayDeque<>();
        queue.offer(new int[]{0, 0, 1});
        visited[0][0] = true;
        
        while (!queue.isEmpty()) {
            int[] curPos = queue.poll();
            int curRow = curPos[0];
            int curCol = curPos[1];
            int curCnt = curPos[2];
            
            if (curRow == rowLength - 1 && curCol == colLength - 1) {
                answer = curCnt;
                return answer;
            }
            
            for (int i = 0; i < 4; i++) {
                int nextRow = curRow + dr[i];
                int nextCol = curCol + dc[i];
                
                if (isValid(nextRow, nextCol, maps)) {
                    queue.offer(new int[]{nextRow, nextCol, curCnt + 1});
                    visited[nextRow][nextCol] = true;
                }
            }
        }
        
        return answer;
    }
}
