#### 문제파악
n = 노드 갯수

vertex = a 노드와 b 노드 사이에 간선이 있다는 의미인 `[a, b]` 배열이 각 행에 들어있는 배열

1번 노드로부터 가장 멀리 떨어진 노드가 몇 개인지 return
#### 접근 방법
주어진 vertex 배열을 일단 **인접 리스트** 형태로 새로 저장한다

가장 멀리 떨어진 노드 = **최단 경로**로 이동했을 때 간선의 개수가 가장 많은 노드들
→ 최단경로니까 bfs 사용

인접 리스트를 가지고 1번 노드를 시작점으로 하여 bfs 노드 간 거리를 각 노드별로 카운트

가장 큰 값 몇 개인지 세어서 Return?
#### 코드 구현
```java
// 구현 시도 코드
import java.util.*;
class Solution {
    Map<Integer, Boolean> visited;
    public int solution(int n, int[][] edge) {
        int cnt = 0;
        
        // 인접 리스트로 변환
        Map<Integer, List<Integer>> graph = new HashMap<>();
        for (int i = 1; i <= n; i++) graph.put(i, new ArrayList<>());
        for (int[] e : edge) {
            graph.get(e[0]).add(e[1]);
            graph.get(e[1]).add(e[0]);
        }
        
        // bfs(1) 호출
        return bfs(1, graph);
                
    }

    public int bfs(int startVertex, Map<Integer, List<Integer>> graph) {
        int cnt = 0;
        Queue<Integer> queue = new LinkedList<>();
        visited = new HashMap<>();
        
        queue.offer(startVertex);
        visited.put(startVertex, true);
        
        while (!queue.isEmpty()) {
            int currVertex = queue.poll(); // 노드만 가져오는게 아닌 노드와 거리 가져왔어야 함
            
            // 거리 비교해서 카운트 증가 및 초기화 필요했음
            cnt = 0;
            for (int nextVertex : graph.get(currVertex)) {
                if (!visited.containsKey(nextVertex)) {
                    cnt++; // 위치가 여기가 아님(이렇게 하면 bfs 실행 횟수세게 됨)

                    queue.offer(nextVertex);
                    visited.put(nextVertex, true);
                }
            }
        }
        
        return cnt;
    }
}
```

```java
import java.util.*;
class Solution {
    boolean[] visited;
    public int solution(int n, int[][] edge) {        
        // 인접 리스트로 변환
        Map<Integer, List<Integer>> graph = new HashMap<>();
        for (int i = 1; i <= n; i++) graph.put(i, new ArrayList<>());
        for (int[] e : edge) {
            graph.get(e[0]).add(e[1]);
            graph.get(e[1]).add(e[0]);
        }
        
        // bfs(1) 호출
        return bfs(1, n, graph);
                
    }

    public int bfs(int startVertex, int n, Map<Integer, List<Integer>> graph) {
        Queue<int[]> queue = new LinkedList<>();
        visited = new boolean[n + 1];
        int count = 0;
        int maxDist = -1;
        
        queue.offer(new int[]{startVertex, 0});
        visited[startVertex] = true;
        
        while (!queue.isEmpty()) {
            int curr[] = queue.poll();
            int currVertex = curr[0];
            int currDist = curr[1];
            
            // 가장 먼 노드 발견할 때마다 count 초기화
            if (maxDist < currDist) {
                maxDist = currDist;
                count = 0;
            }
            count++;
            
            for (int nextVertex : graph.get(currVertex)) {
                if (!visited[nextVertex]) {
                    queue.offer(new int[]{nextVertex, currDist + 1});
                    visited[nextVertex] = true;
                }
            }
        }
        
        return count;
    }
}
```
