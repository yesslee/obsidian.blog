#### 문제파악
n개의 송전탑이 전선을 통해 하나의 트리 형태로 연결되어 있을 때 전선들 중 하나를 끊어서 전력망 네트워크를 2개로 분할

분할된 두 전력망이 갖게되는 송전탑의 개수를 최대한 비슷하게 맞추기

n = 송전탑 개수

wires = 전선 정보 2차원 배열
- `[v1, v2]` = v1 송전탑과 v2 송전탑이 전선으로 연결되어 있다는 것을 의미
- `1 ≤ v1 ≤ v2 ≤ n`

두 전력망으로 나눴을 때 송전탑 개수의 차이(절대값) return
#### 접근 방법
주어진 배열 → 인접 리스트로 변환
노드 중 간선의 개수가 가장 많은 것을 찾는다
간선의 개수가 가장 많은 노드가 여러개라면? → 해당 노드간의 연결을 끊는다

해당 노드를 시작점으로 잡고, 시작점에서 BFS로 갈 수 있는 거리가 가장 긴, 바로 인접한 노드와의 연결을 끊는다?

연결 끊은 뒤 dfs 실행해서 깊이 차이 절대값 구하기?

// 구현 시도 코드
import java.util.*;
class Solution {
    Map<Integer, List<Integer>> graph = new HashMap<>();
    Map<Integer, Boolean> visited = new HashMap<>();
    public int solution(int n, int[][] wires) {
        int answer = -1;
        
        // 인접 리스트로 변환
        for (int i = 1; i <= n; i++) graph.put(i, new ArrayList<>());
        for (int[] edge : wires) {
            graph.get(edge[0]).add(edge[1]);
            graph.get(edge[1]).add(edge[0]);
        }
        
        cutting(n);
        
        System.out.println(graph);
        
        answer = Math.abs(dfs(1, graph) - dfs(n, graph));
        System.out.println(Math.abs(dfs(1, graph) - dfs(n, graph)));
        // System.out.println(dfs(n, graph));
        
        return answer;
    }
    
    public void cutting(int n) {
        // 각 송전탑과 연결된 전선 수 담고있는 리스트
        ArrayList<Integer> edgeCnt = new ArrayList<>();
        edgeCnt.add(0); // 노드 번호 맞추기 위해서 추가
        for (int i = 1; i <= n; i++) {
            edgeCnt.add(graph.get(i).size());
        }
        
        ArrayList<Integer> maxIdx = new ArrayList<>();
        for (int i = 0; i < edgeCnt.size(); i++) {
            if (edgeCnt.get(i) == Collections.max(edgeCnt)) maxIdx.add(i);
        }
        System.out.println(maxIdx);
        
        
        // 간선 개수 가장 큰 값이 2개 이상일 때는 해당 노드 간의 간선을 잘라냄
        if (maxIdx.size() > 1) {
            graph.get(maxIdx.get(0)).remove(maxIdx.get(1));
            graph.get(maxIdx.get(1)).remove(maxIdx.get(0));
        } else {
            // 인접한 노드 중 그 다음으로 노드 개수 많은 걸 잘라냄 -> 이 부분 구현 좀 더 생각하면 제대로 동작할듯
            // 지금은 최대값 노드 이전 노드로 잘라내서 의도대로 동작하는 코드가 아님
            graph.get(Collections.max(edgeCnt) - 1).remove(maxIdx.get(0));
            graph.get(maxIdx.get(0)).remove(Collections.max(edgeCnt) - 1);
        }
    }
    
    public int dfs(int curVertex, Map<Integer, List<Integer>> graph) {
        int cnt = 1;
        visited.put(curVertex, true);
        for (int nextVertex : graph.get(curVertex)) {
            if (!visited.containsKey(nextVertex)) {
                cnt += dfs(nextVertex, graph);
            }
        }
        return cnt;
    }
    
}
