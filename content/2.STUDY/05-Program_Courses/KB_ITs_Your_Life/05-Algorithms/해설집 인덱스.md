```dataviewjs
const sanitize = str => str.replace(/\s/g, "").replace(/[^a-zA-Z가-힣0-9]/g, "");

// 사용자 정의 정렬 우선순위 부여
const typeOrder = {
  "예제": 0,
  "실습": 1,
  "챌린지": 2
};

dv.table(
  ["주차", "유형", "문제", "카테고리", "추천도"],
  dv.pages('"2.STUDY/05-Program_Courses/KB_ITs_Your_Life/05-Algorithms/해설집"')
    .where(p => p.recommend)
    .sort(p => p.file.name, 'asc')
    .sort(p => {
	  const name = p.file.name;
	  const firstChar = name.charAt(0);
	  const isEnglish = /^[A-Za-z]/.test(firstChar);
	  // 영어면 0, 한글 등 다른 문자면 1 → 영어 우선
	  return (isEnglish ? 0 : 1) + name;
	}, 'asc')
	.sort(p => typeOrder[p.type] ?? 99, 'asc')
	.sort(p => p.weeks, 'desc')
    .map(p => [
      (p.weeks ? p.weeks + "주차" : "-"),
      `<span class="type-chip type-${sanitize(p.type)}">${p.type}</span>`,
      p.file.link,
      (Array.isArray(p.category)
        ? p.category.map(c => `<span class="chip chip-${sanitize(c)}">${c}</span>`).join(" ")
        : `<span class="chip chip-${sanitize(p.category ?? "기타")}">${p.category ?? "-"}</span>`),
      "⭐".repeat(Math.round(p.recommend / 20)) //+ ` (${p.recommend}/100)`
    ])
);
```
